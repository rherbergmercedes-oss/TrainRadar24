async function searchStations(inputId, resultId) { const query = document.getElementById(inputId).value.trim(); const area = document.getElementById(resultId); if (query.length < 2) { area.innerHTML = '

Mindestens 2 Buchstaben eingeben.

'; return; } area.innerHTML = '

Suche deutsche Bahnhöfe...

'; const where = "Name LIKE '%" + query.replace(/'/g, "''") + "%' AND (Verkehrsm LIKE '%Bahn%' OR Verkehrsm LIKE '%Zug%')"; const params = new URLSearchParams({ where: where, outFields: "Name,Verkehrsm,Art,Gemeinde,Bundesland,Stop_ID", returnGeometry: "false", resultRecordCount: "100", orderByFields: "Name ASC", f: "json" }); const url = "https://services2.arcgis.com/jUpNdisbWqRpMo35/ArcGIS/rest/services/Haltestelledaten_DELFI/FeatureServer/0/query?" + params.toString(); try { const response = await fetch(url); if (!response.ok) { throw new Error("Datenquelle nicht erreichbar"); } const data = await response.json(); if (data.error) { throw new Error(data.error.message); } const results = data.features || []; const unique = [...new Map( results.map(item => [ item.attributes.Name, item.attributes ]) ).values()]; if (unique.length === 0) { area.innerHTML = '

Kein passender Bahnhof gefunden.

'; return; } area.innerHTML = unique.map(item => `
${item.Name}
${item.Gemeinde || ""} ${item.Bundesland || ""}
${item.Verkehrsm || "Bahnverkehr"}
`).join(""); } catch (error) { area.innerHTML = '

Die Bahnhofsdaten konnten gerade nicht geladen werden. Bitte später erneut versuchen.

'; } }
